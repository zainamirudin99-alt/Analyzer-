function processFile(file) {
  if (file.type !== 'application/pdf') {
    showMsg('uploadMsg', 'error', '❌ Hanya file PDF yang diterima.'); return;
  }
  if (file.size > 20 * 1024 * 1024) {
    showMsg('uploadMsg', 'error', '❌ Ukuran file melebihi 20 MB.'); return;
  }
  currentFile = file;
  dropZone.classList.add('has-file');
  document.getElementById('dzIcon').textContent = '✅';
  document.getElementById('dzText').textContent = file.name;
  const sizeMB = file.size / 1024 / 1024;
  document.getElementById('dzSub').textContent  = `${sizeMB.toFixed(2)} MB`;

  const warn = document.getElementById('pdfSizeWarn');
  if (sizeMB > 10) {
    warn.innerHTML = `<div class="alert alert-error" style="margin-top:9px;"><span class="alert-icon">⚠️</span><span><strong>File besar (${sizeMB.toFixed(1)} MB):</strong> Risiko quota limit (429). Sistem akan retry & beralih model otomatis, namun proses lebih lama. Disarankan kompres PDF ke &lt; 5 MB.</span></div>`;
  } else if (sizeMB > 5) {
    warn.innerHTML = `<div class="alert alert-warn" style="margin-top:9px;"><span class="alert-icon">⚠️</span><span><strong>File cukup besar (${sizeMB.toFixed(1)} MB).</strong> Jika 429, sistem akan otomatis beralih ke model alternatif.</span></div>`;
  } else {
    warn.innerHTML = `<div class="alert alert-success" style="margin-top:9px;"><span class="alert-icon">🌿</span><span>Ukuran file aman (${sizeMB.toFixed(1)} MB) — risiko quota limit kecil.</span></div>`;
  }
  showMsg('uploadMsg', 'success', `📄 File dipilih: ${file.name}`);
}

// ── Main Analysis Flow ──────────────────────────────────────
function startAnalysis() {
  const code  = document.getElementById('companyCode').value.trim().toUpperCase();
  const year  = document.getElementById('fiscalYear').value;
  const notes = document.getElementById('notes').value.trim();

  if (!code)        { showMsg('uploadMsg', 'warn', '⚠️ Kode perusahaan wajib diisi.'); return; }
  if (!year)        { showMsg('uploadMsg', 'warn', '⚠️ Tahun fiskal wajib dipilih.');  return; }
  if (!currentFile) { showMsg('uploadMsg', 'warn', '⚠️ Pilih file PDF terlebih dahulu.'); return; }

  const btn = document.getElementById('btnAnalyze');
  btn.disabled = true;
  btn.innerHTML = '<span class="spinner"></span> Menganalisis…';

  // Hide empty placeholder, prepare score card
  document.getElementById('emptyPreviewCard').style.display = 'none';
  document.getElementById('scorePreviewCard').style.display = 'none';

  setProgress(10, 'Membaca file PDF…');
  showLoading('Membaca PDF…', 'Mengonversi dokumen ke format AI…');

  const reader = new FileReader();
  reader.onload = e => {
    const base64 = e.target.result.split(',')[1];

    setProgress(25, 'Menyimpan data input…');
    setLoading('Menyimpan ke Spreadsheet…', '');

    google.script.run
      .withSuccessHandler(res => {
        if (res.status !== 'ok') { handleError(res.message, btn); return; }

        const rowNumber = res.rowNumber;
        setProgress(45, 'Mengirim PDF ke Gemini AI…');
        setLoading('AI sedang membaca laporan…', 'Jika terkena quota limit, sistem akan retry & beralih model otomatis. Mohon tunggu…');

        google.script.run
          .withSuccessHandler(aRes => {
            hideLoading();
            btn.disabled = false;
            btn.innerHTML = '🔍 Mulai Analisis AI';

            if (aRes.status !== 'ok') {
              setProgress(100, '❌ Gagal');
              document.getElementById('progressFill').style.background = '#ECA8A8';
              const msg = aRes.message || '';
              const is429 = msg.includes('429') || msg.toLowerCase().includes('quota');
              if (is429) {
                showMsg('uploadMsg', 'error',
                  '❌ <strong>Quota Limit (429):</strong> Semua model sudah dicoba. Tunggu beberapa menit lalu coba lagi, atau gunakan PDF yang lebih kecil (&lt; 5 MB).<br><small style="opacity:.75;">' + msg.substring(0, 250) + '</small>'
                );
              } else {
                showMsg('uploadMsg', 'error', '❌ ' + msg);
              }
              document.getElementById('emptyPreviewCard').style.display = '';
              return;
            }

            setProgress(100, '✅ Selesai!', true);

            // Cek fallback model
            google.script.run
              .withSuccessHandler(mRes => {
                let successMsg = `✅ Analisis <strong>${code} ${year}</strong> berhasil disimpan.`;
                if (mRes && mRes.lastUsed && mRes.lastUsed !== mRes.configured) {
                  successMsg += `<br><small style="opacity:.85;">⚠️ Fallback model digunakan: <strong>${mRes.lastUsed}</strong> (${mRes.configured} terkena quota limit).</small>`;
                }
                showMsg('uploadMsg', 'success', successMsg);
              })
              .withFailureHandler(() => {
                showMsg('uploadMsg', 'success', `✅ Analisis <strong>${code} ${year}</strong> berhasil disimpan.`);
              })
              .getLastUsedModel();

            showScorePreview(code, year, aRes.scores);
          })
          .withFailureHandler(err => { hideLoading(); handleError(err.message || String(err), btn); })
          .analyzePDF({ companyCode: code, fiscalYear: year, fileName: currentFile.name, pdfBase64: base64, rowNumber });
      })
      .withFailureHandler(err => { hideLoading(); handleError(err.message || String(err), btn); })
      .saveInputData({ companyCode: code, fiscalYear: year, fileName: currentFile.name, notes });
  };
  reader.onerror = () => { hideLoading(); handleError('Gagal membaca file.', btn); };
  reader.readAsDataURL(currentFile);
}

  }).join('');
}

function renderTable(data) {
  const body = document.getElementById('resultBody');
  if (!data.length) {
    body.innerHTML = `<tr><td colspan="22" style="padding:28px; color:var(--stone); font-style:italic;">📭 Tidak ada data ditemukan.</td></tr>`;
    return;

  body.innerHTML = data.map(row => `

      <td class="td-year">${row.year}</td>

      <td><button class="btn btn-danger btn-sm" onclick="deleteRow('${row.code}','${row.year}')">🗑️</button></td>
    </tr>
  `).join('');
}

function filterTable() {

  renderTable(allResultsData.filter(r =>
    (!code || r.code.includes(code)) && (!year || r.year === year)
  ));
}

function deleteRow(code, year) {
  if (!confirm(`Hapus data ${code} ${year}?`)) return;
  google.script.run
    .withSuccessHandler(res => { if (res.status === 'ok') loadResults(); else showMsg('resultMsg','error','❌ '+res.message); })
    .withFailureHandler(err => showMsg('resultMsg','error','❌ '+(err.message||'Gagal')))
    .deleteResultRow(code, year);
}

function renderSummary(data) {
  const card = document.getElementById('summaryCard');
  if (!data.length) { card.style.display = 'none'; return; }
  card.style.display = 'block';
  const totals = data.map(r => Number(r.total) || 0);
  const avg    = (totals.reduce((s,v) => s+v, 0) / totals.length).toFixed(1);
  const maxV   = Math.max(...totals); const maxRow = data.find(r => Number(r.total) === maxV);
  const minV   = Math.min(...totals); const minRow = data.find(r => Number(r.total) === minV);
  document.getElementById('summaryStats').innerHTML = `
    <div class="stat-box s-green"><div class="stat-val">${data.length}</div><div class="stat-lbl">Jumlah Entri</div></div>
    <div class="stat-box s-teal"><div class="stat-val">${avg}</div><div class="stat-lbl">Rata-rata Skor</div></div>
    <div class="stat-box s-sage"><div class="stat-val">${maxV}</div><div class="stat-lbl">Skor Tertinggi — ${maxRow?.code} ${maxRow?.year}</div></div>
    <div class="stat-box s-earth"><div class="stat-val">${minV}</div><div class="stat-lbl">Skor Terendah — ${minRow?.code} ${minRow?.year}</div></div>
  `;
}

function loadAvailableModels() {
  const btn = document.getElementById('btnLoadModels');
  btn.disabled = true;
  btn.innerHTML = '<span class="spinner spinner-dark"></span> Memuat…';
  showMsg('modelMsg','info','⏳ Mengambil daftar model dari Gemini API…');

  google.script.run
    .withSuccessHandler(res => {
      btn.disabled = false; btn.innerHTML = '🔄 Muat Model';
      if (res.status !== 'ok') { showMsg('modelMsg','error','❌ '+res.message); return; }

      const select = document.getElementById('modelSelect');
      select.innerHTML = ''; select.disabled = false;

      const preferred = ['gemini-2.5-flash','gemini-2.0-flash','gemini-2.5-pro','gemini-2.0-flash-lite'];
      const prefGroup = document.createElement('optgroup');
      prefGroup.label = '⭐ Direkomendasikan';
      let pc = 0;
      preferred.forEach(m => {
        if (res.models.includes(m)) {
          const o = document.createElement('option');
          o.value = m; o.textContent = m;
          if (m === res.currentModel) o.selected = true;
          prefGroup.appendChild(o); pc++;
        }
      });
      if (pc) select.appendChild(prefGroup);

      const otherGroup = document.createElement('optgroup');
      otherGroup.label = '📋 Semua Model';
      res.models.forEach(m => {
        if (!preferred.includes(m)) {
          const o = document.createElement('option');
          o.value = m; o.textContent = m;
          if (m === res.currentModel) o.selected = true;
          otherGroup.appendChild(o);
        }
      });
      if (otherGroup.children.length) select.appendChild(otherGroup);

      if (res.currentModel && !res.models.includes(res.currentModel)) {
        const o = document.createElement('option');
        o.value = res.currentModel; o.textContent = res.currentModel + ' (aktif)'; o.selected = true;
        select.insertBefore(o, select.firstChild);
      }

      document.getElementById('btnSaveModel').disabled = false;
      _showCurrentModel(res.currentModel);
      showMsg('modelMsg','success',`✅ ${res.models.length} model tersedia. Aktif: <strong>${res.currentModel}</strong>`);
    })
    .withFailureHandler(err => {
      btn.disabled = false; btn.innerHTML = '🔄 Muat Model';
      showMsg('modelMsg','error','❌ '+(err.message||'Gagal.'));
    })
    .getAvailableModels();
}

function saveSelectedModel() {
  const chosen = document.getElementById('modelSelect').value;
  if (!chosen) { showMsg('modelMsg','warn','⚠️ Pilih model terlebih dahulu.'); return; }
  const btn = document.getElementById('btnSaveModel');
  btn.disabled = true; btn.innerHTML = '<span class="spinner"></span> Menyimpan…';
  google.script.run
    .withSuccessHandler(res => {
      btn.disabled = false; btn.innerHTML = '💾 Simpan';
      if (res.status === 'ok') { _showCurrentModel(chosen); showMsg('modelMsg','success','✅ '+res.message); }
      else showMsg('modelMsg','error','❌ '+res.message);
    })
    .withFailureHandler(err => { btn.disabled = false; btn.innerHTML = '💾 Simpan'; showMsg('modelMsg','error','❌ '+(err.message||'Gagal.')); })
    .setModel(chosen);
}
